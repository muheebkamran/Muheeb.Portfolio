def bisection_root(x, epsilon=0.01):
    low, high = 0.0, max(1.0, x)
    ans = (high + low) / 2.0
    while abs(ans**2 - x) >= epsilon:
        if ans**2 < x: low = ans
        else: high = ans
        ans = (high + low) / 2.0
    return ans

#include <stdio.h>

int main(void) {
    int arr[] = {12, 45, 78, 23, 90};
    int n = sizeof(arr) / sizeof(arr[0]);
    int max = arr[0];
    for (int i = 1; i < n; i++) {
        if (arr[i] > max) max = arr[i];
    }
    printf("Max: %d\n", max);
    return 0;
}
